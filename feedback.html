export default async function handler(req, res) { if (req.method !== 'POST') { return res.status(405).json({ error: 'Method not allowed' }); } const { examKey, correctCount, totalQuestions, questions } = req.body; if (!questions || !Array.isArray(questions)) { return res.status(400).json({ error: 'Некорректные данные теста' }); } // Расчет баллов TOPIK const isTopik2 = examKey && String(examKey).toLowerCase().includes('topik2'); const maxScale = isTopik2 ? 300 : 200; const percent = totalQuestions > 0 ? correctCount / totalQuestions : 0; const finalScore = Math.round(percent * maxScale); let grade = 'Не сдан'; let passed = false; if (isTopik2) { if (finalScore >= 230) { grade = '6 급'; passed = true; } else if (finalScore >= 190) { grade = '5 급'; passed = true; } else if (finalScore >= 150) { grade = '4 급'; passed = true; } else if (finalScore >= 120) { grade = '3 급'; passed = true; } } else { if (finalScore >= 140) { grade = '2 급'; passed = true; } else if (finalScore >= 80) { grade = '1 급'; passed = true; } } // Отбираем только ошибки const mistakes = questions .map((q, index) => ({ index: index + 1, question: q.title || q.question || q.text || `Вопрос ${index + 1}`, userAnswer: q.userAnswer ?? q.selectedAnswer ?? 'Не выбран', correctAnswer: q.correctAnswer ?? q.rightAnswer ?? 'Нет ответа', isCorrect: q.isCorrect !== undefined ? q.isCorrect : (q.userAnswer === q.correctAnswer) })) .filter(q => !q.isCorrect); // Если нет ошибок — возвращаем готовый ответ без вызова ИИ if (mistakes.length === 0) { return res.status(200).json({ finalScore, maxScale, grade, passed, overallFeedback: 'Отличная работа! Вы решили весь тест без единой ошибки. Отличный уровень подготовки!', questionFeedback: [] }); } // Формируем системную роль и промпт const systemMessage = `Ты — старший преподаватель корейского языка и официальный экзаменатор TOPIK. Твоя задача — дать глубокий, дружелюбный и методически точный разбор ошибок студента. Отвечай СТРОГО в формате JSON (без markdown, без \`\`\`) со структурой: { "overallFeedback": "текст с полным анализом результата и итогом — сдал студент экзамен или нет", "questionFeedback": [ { "index": номер_вопроса, "explanation": "разбор ошибки" } ] }`; const userMessage = `Экзамен: ${isTopik2 ? 'TOPIK II' : 'TOPIK I'} Результат: ${correctCount} из ${totalQuestions} (${finalScore}/${maxScale} баллов) Присвоенный гып: ${grade} Статус: ${passed ? 'Тест сдан' : 'Тест не сдан'} Ошибки студента: ${JSON.stringify(mistakes.map(m => ({ номер: m.index, вопрос: m.question, ответ_студента: m.userAnswer, правильный_ответ: m.correctAnswer })), null, 2)} Требования к разбору: 1. "overallFeedback": Дай развернутый анализ (3-4 предложения). Похвали за старание, укажи слабую сторону (грамматика, лексика, чтение), скажи прямо сдал ли студент тест и дай 1-2 конкретных совета, что выучить для следующего уровня. 2. "questionFeedback": Для КАЖДОЙ ошибки объясни понятным языком: почему ответ студента не подходит (разбери грамматику/смысл) и почему правильный вариант верен. Поясняй значение корейских грамматических конструкций и слов, если они есть.`; try { const response = await fetch('https://api.openai.com/v1/chat/completions', { method: 'POST', headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${process.env.OPENAI_API_KEY}` }, body: JSON.stringify({ model: 'gpt-4o-mini', response_format: { type: 'json_object' }, // Аппаратный JSON-режим OpenAI temperature: 0.4, messages: [ { role: 'system', content: systemMessage }, { role: 'user', content: userMessage } ] }) }); if (!response.ok) { const errBody = await response.text(); throw new Error(`OpenAI API error (${response.status}): ${errBody}`); } const data = await response.json(); const rawContent = data?.choices?.[0]?.message?.content; if (!rawContent) { throw new Error('Пустой ответ от модели'); } let parsedData; try { parsedData = JSON.parse(rawContent); } catch (parseErr) { throw new Error(`Не удалось распарсить JSON от модели: ${parseErr.message}`); } return res.status(200).json({ finalScore, maxScale, grade, passed, overallFeedback: parsedData.overallFeedback || 'Не удалось сгенерировать общий фидбек.', questionFeedback: Array.isArray(parsedData.questionFeedback) ? parsedData.questionFeedback : [] }); } catch (err) { console.error('AI error:', err); return res.status(500).json({ error: 'Ошибка генерации фидбека', details: err.message, finalScore, maxScale, grade, passed, overallFeedback: null, questionFeedback: [] }); } }
